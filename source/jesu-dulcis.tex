\bigskip

\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{2.} 
Nil cánitur suávius,\\
Nil audítur jucúndius,\\
Nil cogitátur dúlcius,\\
Quam Jesus Dei Fílius.
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{2.} No voice can sing, no heart can
frame,\\
Nor can the memory find,\\
A sweeter sound than Thy blest Name,\\
O Saviour of mankind.
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{3.} 
Jesu spes pæniténtibus,\\
Quam pius es peténtibus!\\
Quam bonus te quæréntibus!\\
Sed quid inveniéntibus?
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{3.} O hope of every contrite heart,\\
O joy of all the meek,\\
To those who fall, how kind Thou art!\\
How good to those who seek.
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{4.} 
Nec lingua valet dícere,\\
Nec líttera exprímere:\\
Expértus potest crédere,\\
Quid sit Jesum dilígere.
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{4.} But what to those that find? Ah! this\\
No tongue nor pen can show;\\
The love of Jesus, what it is,\\
None but His loved ones know.
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{5.} 
Sis Jesu nostrum gáudium,\\
Qui es futúrus prǽmium:\\
Sit nostra in te glória,\\
Per cuncta semper sǽcula.
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{5.} Jesu, our only joy be Thou,\\
As Thou our prize wilt be,\\
Jesu, be Thou our glory now,\\
And through eternity.
\end{minipage}

\bigskip

\mysource{Attributed to St. Bernard of Clairvaux, 1091--1153\\
Translated by Fr. Edward Caswall, 1814--78}
