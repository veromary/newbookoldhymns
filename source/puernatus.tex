
\smallskip

\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{2.} 
Assúmpsit carnem Fílius,\\
Dei Patris altíssimus,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{2.} 
God the Father’s Son did take,\\
A human nature for our sake,
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{3.} Per Gabriélem núntium,\\
Virgo concépit Fílium,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{3.} Through Gabriel’s message,\\
A Virgin hath conceived a Son,
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{4.} 
Tamquam sponsus de
thálamo,\\
Procéssit Matris útero,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{4.} Like a bridegroom from his home,\\
Proceeding from His Mother’s
womb,
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{5.} 
Hic jacet in præsépio,\\
Qui regnat sine término,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{5.} He lies within a manger poor,\\
Who lives and reigns for ever more,
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{6.} 
Et Angelus pastóribus,\\
Revélat quod sit Dóminus,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{6.} 
And Angels to the shepherds say,\\
Revealing where our Lord lay,
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{7.} 
Reges de Saba véniunt,\\
Aurum, thus, myrrham
ófferunt,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{7.} From Saba come the eastern Kings,\\
Gold, myrrh and incense offering,
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{8.} 
Intrántes domum ínvicem,\\
Novum salútant Príncipem,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{8.} They enter in the dwelling-place,\\
To hail the new-born Prince of
Peace,
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{9.} 
In hoc natáli gáudio,\\
Benedicámus Dómino,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{9.} There is joy on this birthday,\\
Let us bless the Lord,
\end{minipage}

\bigskip


\noindent\begin{minipage}{0.48\textwidth}
\hskip-3pt{10.} 
Laudétur sancta Trínitas,\\
Deo dicámus grátias,
\end{minipage}%
\hskip5pt
\begin{minipage}{0.55\textwidth}
\itshape
\hskip-3pt{10.} The Holy Trinity be praised,\\
Let us give thanks unto our God,
\end{minipage}

\bigskip

\mysource{Gesangbuch, 1548}
